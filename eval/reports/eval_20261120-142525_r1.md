SELECT p.Category, p.ProductName, SUM(oi.Quantity * oi.UnitPrice) AS Ciro FROM Products AS p JOIN OrderItems AS oi ON p.ProductID = oi.ProductID GROUP BY p.Category, p.ProductID ORDER BY p.Category, Ciro DESC
